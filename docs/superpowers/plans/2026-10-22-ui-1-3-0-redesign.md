# UI验收台 1.3.0 主窗口改版实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把主窗口改成「视觉无窗口」的手机形态：手机屏就是画布，左侧是常驻轨道 + 悬浮抽屉坞，运行/截图收在手机下方的胶囊里。

**Architecture:** 底层仍是标准 `NSWindow`（隐藏标题栏和系统交通灯、透明背景），内容由左往右是 `DockColumn`（72pt 轨道 + 300pt 抽屉槽）和 `PhoneStage`（手机框 + 胶囊）。纯状态与几何计算放进 `UITestBenchCore` 并用 XCTest 驱动；App 层只做 SwiftUI 视图和 AppKit 窗口配置。镜像链路、坐标映射、scrcpy 协议一律不动。

**Tech Stack:** Swift 5.9 / SwiftUI / AppKit / XcodeGen；部署目标 macOS 12.3；只用系统字体与 SF Symbols；零第三方依赖。

```swift
import CoreGraphics
import Foundation

/// 手机在窗口里占多大、圆角多少。纯几何，不碰视图。
public struct PhoneLayout: Equatable, Sendable {
    public let screenSize: CGSize
    public let cornerRadius: CGFloat
    public let bezel: CGFloat

    public init(screenSize: CGSize, cornerRadius: CGFloat, bezel: CGFloat) {
        self.screenSize = screenSize
        self.cornerRadius = cornerRadius
        self.bezel = bezel
    }
}

public enum PhoneLayoutCalculator {
    public static let maxScreenWidth: CGFloat = 440
    public static let verticalPadding: CGFloat = 24
    public static let capsuleHeight: CGFloat = 48
    public static let capsuleGap: CGFloat = 12
    public static let bezel: CGFloat = 7
    public static let cornerRadiusRatio: CGFloat = 0.056

    public static func layout(available: CGSize, profile: DeviceProfile) -> PhoneLayout {
        // 屏幕左右各有一条边框，可用宽度里先把它们扣掉，再与设计上限取小。
        let widthBudget = min(maxScreenWidth, max(0, available.width - bezel * 2))
        guard available.width > 0, available.height > 0,
              profile.width > 0, profile.height > 0,
              widthBudget > 0
        else {
            return PhoneLayout(screenSize: .zero, cornerRadius: 0, bezel: bezel)
        }

        // 胶囊、间距、上下留白和边框都要先扣掉，剩下的才是屏幕能用的高度。
        let usableHeight = max(
            0,
            available.height
                - verticalPadding * 2
                - capsuleHeight
                - capsuleGap
                - bezel * 2
        )
        let heightScale = usableHeight / CGFloat(profile.height)
        let widthScale = widthBudget / CGFloat(profile.width)
        let scale = min(heightScale, widthScale)

        let screenWidth = floored(CGFloat(profile.width) * scale)
        let screenHeight = floored(CGFloat(profile.height) * scale)
        return PhoneLayout(
            screenSize: CGSize(width: screenWidth, height: screenHeight),
            cornerRadius: screenWidth * cornerRadiusRatio,
            bezel: bezel
        )
    }

    /// 向下取整前先抹掉浮点乘除留下的一丝误差（如 439.99999999999994）：
    /// 宁可少半个点，也不能把手机挤出窗口。
    private static func floored(_ value: CGFloat) -> CGFloat {
        (value + 1e-9).rounded(.down)
    }
}
```

```swift
import Foundation

/// 只做浅色/深色两态，不跟随系统。
public enum ThemeMode: String, CaseIterable, Equatable, Sendable {
    case light
    case dark

    public static let defaultMode: ThemeMode = .light
    public static let userDefaultsKey = "UITestBench.themeMode"
}

/// 读写持久化的主题；缺省或无法识别时回退到浅色。
public struct ThemePreference {
    private let defaults: UserDefaults

    public init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
    }

    public var mode: ThemeMode {
        get {
            guard let raw = defaults.string(forKey: ThemeMode.userDefaultsKey),
                  let mode = ThemeMode(rawValue: raw)
            else {
                return .defaultMode
            }
            return mode
        }
        nonmutating set {
            defaults.set(newValue.rawValue, forKey: ThemeMode.userDefaultsKey)
        }
    }
}
```

```swift
import Foundation

public enum DockModule: String, CaseIterable, Equatable, Sendable {
    case material
    case device
    case app
    case diagnostics
    case settings
}

/// 坞的展示规则：钉住的优先于 hover 的，同一时间只有一个抽屉。
public struct DockDrawerState: Equatable, Sendable {
    public private(set) var hovered: DockModule?
    public private(set) var pinned: DockModule?

    public init() {}

    public var presented: DockModule? { pinned ?? hovered }

    public mutating func hover(_ module: DockModule) {
        guard pinned == nil else { return }
        hovered = module
    }

    public mutating func unhover(_ module: DockModule) {
        guard hovered == module else { return }
        hovered = nil
    }

    /// 同一模块再点一次取消钉住；点别的模块直接换抽屉。
    public mutating func pin(_ module: DockModule) {
        pinned = pinned == module ? nil : module
        hovered = nil
    }

    /// 只在整体收起时用；按模块的收起计时请用 `unhover(_:)`，避免误关别的抽屉。
    public mutating func clearHover() {
        hovered = nil
    }

    public mutating func dismiss() {
        pinned = nil
        hovered = nil
    }
}
```

```swift
public enum StandbyScreenState: Equatable, Sendable {
    case starting
    case failed(String)
    case needsAPK
    case readyToStart
    /// 设备已经起来，但画面流还没接上。
    case connecting
}

public struct StandbyScreenInput: Equatable, Sendable {
    /// 设备正在启动或停止；两种过渡都先显示进度。
    public let isStarting: Bool
    /// 当前失败原因，设备运行时的画面失败也算；失败清除后必须传 nil。
    public let failure: String?
    /// 是否已经拿到可运行的 APK；不兼容的包应传 false。
    public let hasAPK: Bool
    /// 模拟器是否已经进入运行状态（启动中不算）。
    public let deviceRunning: Bool

    public init(isStarting: Bool, failure: String?, hasAPK: Bool, deviceRunning: Bool) {
        self.isStarting = isStarting
        self.failure = failure
        self.hasAPK = hasAPK
        self.deviceRunning = deviceRunning
    }
}

/// 优先级：启动/停止中 > 当前失败 > 设备已运行（等画面）> 有没有可运行的 APK。
public enum StandbyScreenStateResolver {
    public static func resolve(_ input: StandbyScreenInput) -> StandbyScreenState {
        if input.isStarting { return .starting }
        if let failure = input.failure { return .failed(failure) }
        if input.deviceRunning { return .connecting }
        return input.hasAPK ? .readyToStart : .needsAPK
    }
}
```

```swift
import AppKit
import SwiftUI
import UITestBenchCore

@MainActor
final class DockModel: ObservableObject {
    /// 先做「点击钉住」。想退回纯 hover，把这里改成 false 即可。
    static let supportsPinning = true

    static let hoverDelay: TimeInterval = 0.15
    static let closeDelay: TimeInterval = 0.25

    @Published private(set) var state = DockDrawerState()

    private var hoverWork: DispatchWorkItem?
    private var pendingHover: DockModule?
    private var closeWork: DispatchWorkItem?
    private var outsideMonitor: Any?

    var presented: DockModule? { state.presented }
    var pinned: DockModule? { state.pinned }

    func hover(_ module: DockModule, inside: Bool) {
        if inside {
            closeWork?.cancel()
            closeWork = nil
            guard state.pinned == nil else { return }
            hoverWork?.cancel()
            pendingHover = module
            let work = DispatchWorkItem { [weak self] in
                self?.pendingHover = nil
                self?.state.hover(module)
            }
            hoverWork = work
            DispatchQueue.main.asyncAfter(deadline: .now() + Self.hoverDelay, execute: work)
        } else {
            // 鼠标从 A 划到 B 时，A 的离开事件可能晚于 B 的进入事件。
            // 只清理属于 A 的计时，否则会把 B 刚排好的展开也取消掉。
            if pendingHover == module {
                hoverWork?.cancel()
                hoverWork = nil
                pendingHover = nil
            }
            guard state.pinned == nil else { return }
            if state.hovered == module {
                scheduleClose()
            }
        }
    }

    /// 鼠标进入抽屉时取消收起；离开时只有未钉住才收起。
    func drawerHover(_ inside: Bool) {
        if inside {
            closeWork?.cancel()
            closeWork = nil
        } else if state.pinned == nil {
            scheduleClose()
        }
    }

    func togglePin(_ module: DockModule) {
        hoverWork?.cancel()
        hoverWork = nil
        pendingHover = nil
        if Self.supportsPinning {
            state.pin(module)
        } else {
            state.hover(module)
        }
    }

    func dismiss() {
        hoverWork?.cancel()
        closeWork?.cancel()
        hoverWork = nil
        closeWork = nil
        pendingHover = nil
        state.dismiss()
    }

    /// 钉住后点窗口其他位置取消。坞列固定从窗口左边缘算 372pt。
    func startOutsideClickMonitor(dockWidth: CGFloat) {
        guard outsideMonitor == nil else { return }
        outsideMonitor = NSEvent.addLocalMonitorForEvents(matching: .leftMouseDown) { [weak self] event in
            guard let self, self.state.pinned != nil else { return event }
            if event.locationInWindow.x > dockWidth {
                self.dismiss()
            }
            return event
        }
    }

    func stopOutsideClickMonitor() {
        if let outsideMonitor {
            NSEvent.removeMonitor(outsideMonitor)
        }
        outsideMonitor = nil
    }

    private func scheduleClose() {
        closeWork?.cancel()
        let work = DispatchWorkItem { [weak self] in
            self?.state.clearHover()
        }
        closeWork = work
        DispatchQueue.main.asyncAfter(deadline: .now() + Self.closeDelay, execute: work)
    }
}
```

```swift
    private enum StepState {
        case pending
        case active
        case done
    }

    private var steps: [(title: String, state: StepState)] {
        switch model.phase {
        case .idle, .failed:
            return [("组件清单", .pending), ("下载与安装", .pending), ("完成", .pending)]
        case .preparing:
            return [("组件清单", .active), ("下载与安装", .pending), ("完成", .pending)]
        case .installing:
            return [("组件清单", .done), ("下载与安装", .active), ("完成", .pending)]
        case .ready:
            return [("组件清单", .done), ("下载与安装", .done), ("完成", .done)]

    private func stepRow(title: String, state: StepState) -> some View {

            Text(title)
                .font(.caption)
                .foregroundStyle(state == .pending ? palette.textTertiary : palette.textPrimary)

        return "开始准备"

    private var hasValidMirrorURLs: Bool {

              let systemImage = URL(string: systemImageURLText)
        else { return false }
        return repository.scheme?.hasPrefix("http") == true
            && systemImage.scheme?.hasPrefix("http") == true
    }

    private func applyMirrorSettings() {
        guard let repository = URL(string: repositoryURLText),
              let systemImage = URL(string: systemImageURLText)
        else { return }

        let current = model.configuration
        model.configuration = SdkBootstrapConfiguration(
            repositoryURL: repository,
            systemImageRepositoryURL: systemImage,
            systemImagePackagePath: current.systemImagePackagePath,
            hostOS: current.hostOS,
            hostArch: current.hostArch
        )
        showsMirrorSettings = false
    }

    private func importOfflineFolder() {
        let panel = NSOpenPanel()
        panel.canChooseFiles = false
        panel.canChooseDirectories = true
        panel.allowsMultipleSelection = false
        panel.prompt = "导入"
        panel.message = "选择存放平台工具、模拟器和系统镜像压缩包的文件夹"
        guard panel.runModal() == .OK, let folder = panel.url else { return }
        model.start(offlineFolder: folder)
    }
}
```

## 3. 风险与回退

| 风险 | 表现 | 回退 |
|---|---|---|
| 透明窗口阴影不跟随内容 | 桌面上出现矩形黑影 | `window.hasShadow = false`，在 PhoneFrame/Drawer 上保留 SwiftUI 远影；窗口四周加 24pt 透明边 |
| 隐藏系统交通灯后全屏/最小化异常 | 绿点无效、全屏黑边 | 回到 Q1 的方案 B：保留系统交通灯，坞顶只留 28pt 透明带 |
| SwiftUI 裁剪裁不掉 NSView | 手机屏幕四角是直角 | 在 `MirrorDisplayView` 上设置 `layer.cornerRadius` 与 `masksToBounds`；坐标映射不动 |
| hover 抖 | 鼠标在轨道与抽屉之间反复进出时闪烁 | `DockModel.supportsPinning = false` 退回纯 hover；或把 `closeDelay` 提到 0.35s |
| 点击外部收起误伤镜像 | 点手机时抽屉收起同时设备也收到点击 | 这是预期行为；若验收觉得干扰，把外部点击监视改成只响应坞列之外的「非手机」区域 |
| 删除键不再退格 | Android 输入框里按 Delete 无反应 | 产品决定；回退只需恢复 `KeyboardInput` 的 `.key(.back)` 分支与对应测试 |
| 手动二态主题与系统不一致 | 取景卡片、关于页颜色不对 | 已用 `applyTheme(dark:)` 和 `.benchTheme` 覆盖；仍有遗漏时按 Task 25–27 补 |

## 4. 不做的事

- 不做多窗口、多设备。
- 不做手机框的刘海/侧键/摄像头。
- 不做转屏时手机框旋转；横屏内容仍按 `resizeAspect` letterbox。
- 不做坞的拖拽排序、自定义模块。
- 不改 scrcpy、解码、坐标映射、设备 ID 逻辑。
- 不把截图或镜像内容写盘（除原有截图保存行为）。
